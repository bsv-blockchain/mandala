## Review Focus

1. **A token-shaped output encoded non-canonically** (e.g. amount 5 as `01 05`, id via `PUSHDATA1`, amount `00`) is refused as `ERR_SHAPE`, never silently skipped. Pinned in Task 2 (codec) and Task 10 (manager).
2. **A CBOR payload in a shape another decoder accepts** (float `1.0`, tag 42, a negative int, an array, unsorted keys, depth 5, trailing byte) carries no attributes, so a deploy or committed authority built on it is refused. Pinned in Task 1 and Task 10.
3. **A spend whose owner row is missing** (a lookup write that failed earlier) is repaired inline and admitted. When nothing can repair it the answer is the infra reject (`ERR_UNAVAILABLE`), never `ERR_SHAPE` / `ERR_LINKAGE`. Pinned in Task 9 and Task 10.
4. **Two authority outputs of one token in one tx, only one committed;** split and combine with no commitment and Δ = 0 are admitted. Spending an authority without creating one is refused (`ERR_AUTHORITY`). Pinned in Task 9.
5. **A replayed deploy** (same pkh and linkage, different txid) without a valid `deploySig` is refused (`ERR_AUTHORITY`). An untrusted owner gets `ERR_UNTRUSTED`, which is never persisted. Pinned in Task 9 and Task 10.

```ts
// Strict DAG-CBOR subset used by Mandala (BRC-162 spec §3.5). Hand-rolled on
// purpose: one whitelist that the TS and Go overlays implement byte for byte.
// Map keys are text and sorted by encoded bytes, which for text keys equals
// DAG-CBOR's length-first order.

export type StrictCborValue = bigint | Uint8Array | string | null | boolean | StrictCborMap
export interface StrictCborMap { readonly [key: string]: StrictCborValue }
export type StrictCborInput = StrictCborValue | number

export const STRICT_CBOR_MAX_BYTES = 4096
export const STRICT_CBOR_MAX_DEPTH = 4
const MAX_UINT64 = (1n << 64n) - 1n

export class StrictCborError extends Error {
  constructor (message: string) {
    super(message)
    this.name = 'StrictCborError'
  }
}

const fail = (message: string): never => { throw new StrictCborError(message) }
const utf8 = new TextEncoder()

const header = (major: number, value: bigint): number[] => {
  const m = major << 5
  if (value < 24n) return [m | Number(value)]
  if (value < 0x100n) return [m | 24, Number(value)]
  if (value < 0x10000n) return [m | 25, Number(value >> 8n), Number(value & 0xffn)]
  if (value < 0x100000000n) return [m | 26, ...bigEndian(value, 4)]
  return [m | 27, ...bigEndian(value, 8)]
}

const bigEndian = (value: bigint, size: number): number[] => {
  const out = new Array<number>(size)
  for (let i = size - 1; i >= 0; i--) { out[i] = Number(value & 0xffn); value >>= 8n }
  return out
}

const encodeKey = (key: string): number[] => {
  const bytes = Array.from(utf8.encode(key))
  return [...header(3, BigInt(bytes.length)), ...bytes]
}

const compareBytes = (a: readonly number[], b: readonly number[]): number => {
  const n = Math.min(a.length, b.length)
  for (let i = 0; i < n; i++) if (a[i] !== b[i]) return a[i] - b[i]
  return a.length - b.length
}

const encodeValue = (value: StrictCborInput, depth: number): number[] => {
  if (typeof value === 'bigint') {
    if (value < 0n || value > MAX_UINT64) fail('integer outside 0..2^64-1')
    return header(0, value)
  }
  if (typeof value === 'number') {
    if (!Number.isSafeInteger(value) || value < 0) fail('number must be a safe non-negative integer')
    return header(0, BigInt(value))
  }
  if (typeof value === 'string') {
    const bytes = Array.from(utf8.encode(value))
    return [...header(3, BigInt(bytes.length)), ...bytes]
  }
  if (value === null) return [0xf6]
  if (value === true) return [0xf5]
  if (value === false) return [0xf4]
  if (value instanceof Uint8Array) return [...header(2, BigInt(value.length)), ...value]
  if (typeof value === 'object') return encodeMap(value as Record<string, StrictCborInput>, depth + 1)
  return fail('unsupported value type')
}

const encodeMap = (map: Record<string, StrictCborInput>, depth: number): number[] => {
  if (depth > STRICT_CBOR_MAX_DEPTH) fail('map nesting deeper than 4')
  const entries = Object.entries(map).map(([k, v]) => ({ key: encodeKey(k), value: v }))
  entries.sort((a, b) => compareBytes(a.key, b.key))
  const out = header(5, BigInt(entries.length))
  for (const e of entries) out.push(...e.key, ...encodeValue(e.value, depth))
  return out
}

export function encodeStrictCbor (map: Record<string, StrictCborInput>): number[] {
  if (map === null || typeof map !== 'object' || map instanceof Uint8Array) fail('top level must be a map')
  const out = encodeMap(map, 1)
  if (out.length > STRICT_CBOR_MAX_BYTES) fail('encoding exceeds 4096 bytes')
  return out
}

class Reader {
  pos = 0
  constructor (readonly bytes: Uint8Array) {}
  byte (): number {
    if (this.pos >= this.bytes.length) fail('truncated input')
    return this.bytes[this.pos++]
  }
  take (n: number): Uint8Array {
    if (this.pos + n > this.bytes.length) fail('truncated input')
    const out = this.bytes.subarray(this.pos, this.pos + n)
    this.pos += n
    return out
  }
  // Returns [major, value]; rejects indefinite/reserved and non-minimal headers.
  head (): [number, bigint] {
    const b = this.byte()
    const major = b >> 5
    const info = b & 0x1f
    if (info < 24) return [major, BigInt(info)]
    if (info > 27) fail(info === 31 ? 'indefinite length' : 'reserved additional info')
    const size = 1 << (info - 24)
    let v = 0n
    for (const x of this.take(size)) v = (v << 8n) | BigInt(x)
    const min = [24n, 0x100n, 0x10000n, 0x100000000n][info - 24]
    if (v < min) fail('non-minimal header')
    return [major, v]
  }
}

const length = (v: bigint, r: Reader): number => {
  if (v > BigInt(r.bytes.length)) fail('length exceeds input')
  return Number(v)
}

const decodeValue = (r: Reader, depth: number): StrictCborValue => {
  const [major, v] = r.head()
  switch (major) {
    case 0: return v
    case 2: return Uint8Array.from(r.take(length(v, r)))
    case 3: return decodeText(r.take(length(v, r)))
    case 5: return decodeMapBody(r, length(v, r), depth + 1)
    case 7: return decodeSimple(v)
    default: return fail(`major type ${major} not allowed`)
  }
}

// Hand-rolled strict UTF-8 validation (RFC 3629): no overlongs, no surrogates,
// max U+10FFFF. Platform-independent (Hermes/RN) and mirrored exactly in Go.
const validUtf8 = (b: Uint8Array): boolean => {
  let i = 0
  while (i < b.length) {
    const c = b[i]
    if (c < 0x80) { i += 1; continue }
    let need: number; let min: number; let cp: number
    if (c >= 0xc2 && c <= 0xdf) { need = 1; min = 0x80; cp = c & 0x1f } else if (c >= 0xe0 && c <= 0xef) { need = 2; min = 0x800; cp = c & 0x0f } else if (c >= 0xf0 && c <= 0xf4) { need = 3; min = 0x10000; cp = c & 0x07 } else return false
    if (i + need >= b.length + 0 && i + need > b.length - 1 + 1) return false
    for (let k = 1; k <= need; k++) {
      const x = b[i + k]
      if (x === undefined || (x & 0xc0) !== 0x80) return false
      cp = (cp << 6) | (x & 0x3f)
    }
    if (cp < min || cp > 0x10ffff || (cp >= 0xd800 && cp <= 0xdfff)) return false
    i += need + 1
  }
  return true
}

const decodeText = (bytes: Uint8Array): string => {
  if (!validUtf8(bytes)) fail('invalid UTF-8')
  return new TextDecoder('utf-8').decode(bytes)
}

const decodeSimple = (v: bigint): StrictCborValue => {
  if (v === 20n) return false
  if (v === 21n) return true
  if (v === 22n) return null
  return fail('simple value or float not allowed')
}

const decodeMapBody = (r: Reader, count: number, depth: number): StrictCborMap => {
  if (depth > STRICT_CBOR_MAX_DEPTH) fail('map nesting deeper than 4')
  const out: Record<string, StrictCborValue> = {}
  let previous: number[] | undefined
  for (let i = 0; i < count; i++) {
    const start = r.pos
    const [major, len] = r.head()
    if (major !== 3) fail('map key must be text')
    const key = decodeText(r.take(length(len, r)))
    const encodedKey = Array.from(r.bytes.subarray(start, r.pos))
    if (previous !== undefined && compareBytes(previous, encodedKey) >= 0) fail('map keys unsorted or duplicated')
    previous = encodedKey
    out[key] = decodeValue(r, depth)
  }
  return out
}

export function decodeStrictCbor (input: readonly number[] | Uint8Array): StrictCborMap {
  const bytes = input instanceof Uint8Array ? input : Uint8Array.from(input)
  if (bytes.length > STRICT_CBOR_MAX_BYTES) fail('input exceeds 4096 bytes')
  const r = new Reader(bytes)
  const [major, count] = r.head()
  if (major !== 5) fail('top level must be a map')
  const map = decodeMapBody(r, length(count, r), 1)
  if (r.pos !== bytes.length) fail('trailing bytes')
  const again = encodeStrictCbor(map)
  if (again.length !== bytes.length || compareBytes(again, Array.from(bytes)) !== 0) fail('non-canonical encoding')
  return map
}

export function tryDecodeStrictCbor (input: readonly number[] | Uint8Array): StrictCborMap | undefined {
  try { return decodeStrictCbor(input) } catch (e) {
    if (e instanceof StrictCborError) return undefined
    throw e
  }
}
```

```ts
// BRC-162 (BSV-21 binary) token-output codec with Mandala's push canonicality
// (spec §3.1): every value has exactly one accepted encoding.
import { LockingScript, OP, UnlockingScript } from '@bsv/sdk/script'
import type ScriptChunk from '@bsv/sdk/script/ScriptChunk'
import type { ScriptTemplate, ScriptTemplateUnlock } from '@bsv/sdk/script'
import { hash160 } from '@bsv/sdk/primitives/Hash'
import { toArray, toHex } from '@bsv/sdk/primitives/utils'
// P2PKH, PrivateKey, WalletInterface, WalletProtocol, WalletCounterparty: import by name from their barrels.

export const BSV21_MAX_AMOUNT = (1n << 64n) - 1n
export type Bsv21Role = 'deploy' | 'authority' | 'value'
export class Bsv21BinaryError extends Error {
  constructor (message: string) { super(message); this.name = 'Bsv21BinaryError' }
}
const fail = (m: string): never => { throw new Bsv21BinaryError(m) }

const isPushOp = (op: number): boolean => op <= OP.OP_PUSHDATA4 || op === OP.OP_1NEGATE || (op >= OP.OP_1 && op <= OP.OP_16)

export function encodeAmountChunk (amount: bigint): ScriptChunk {
  if (amount < 0n || amount > BSV21_MAX_AMOUNT) fail('amount outside 0..2^64-1')
  if (amount === 0n) return { op: OP.OP_0 }
  if (amount <= 16n) return { op: OP.OP_1 + Number(amount) - 1 }
  const data: number[] = []
  let v = amount
  while (v > 0n) { data.push(Number(v & 0xffn)); v >>= 8n }
  if ((data[data.length - 1] & 0x80) !== 0) data.push(0x00)
  return { op: data.length, data }
}

export function decodeAmountChunk (chunk: ScriptChunk): bigint {
  const { op, data } = chunk
  if (op === OP.OP_0) return 0n
  if (op >= OP.OP_1 && op <= OP.OP_16) return BigInt(op - OP.OP_1 + 1)
  if (op < 0x01 || op > 0x09 || data === undefined || data.length !== op) return fail('amount must be OP_0, OP_1..OP_16 or a direct push of 1-9 bytes')
  const last = data[data.length - 1]
  if ((last & 0x80) !== 0) fail('amount must not be negative')
  if ((last & 0x7f) === 0 && (data.length === 1 || (data[data.length - 2] & 0x80) === 0)) fail('amount is not minimally encoded')
  let v = 0n
  for (let i = data.length - 1; i >= 0; i--) v = (v << 8n) | BigInt(data[i])
  if (v <= 16n) fail('amounts 0..16 must use OP_0/OP_1..OP_16')
  if (v > BSV21_MAX_AMOUNT) fail('amount exceeds 2^64-1')
  return v
}

const TOKEN_ID_RE = /^[0-9a-f]{64}_0$/
export function tokenIdFromString (id: string): number[] {
  if (!TOKEN_ID_RE.test(id)) fail('token id must be <64 lowercase hex>_0')
  return toArray(id.slice(0, 64), 'hex').reverse()
}
export function tokenIdToString (tokenId: readonly number[]): string {
  if (tokenId.length !== 32) fail('token id must be 32 bytes')
  return `${toHex([...tokenId].reverse())}_0`
}

export function isTokenShaped (script: LockingScript): boolean {
  const c = script.chunks
  return c.length >= 3 && isPushOp(c[0].op) && isPushOp(c[1].op) && c[2].op === OP.OP_2DROP
}

// Minimal push opcode for raw payload bytes (MINIMALDATA).
const canonicalPushOp = (data: readonly number[]): number => {
  if (data.length === 0) return OP.OP_0
  if (data.length === 1 && data[0] >= 1 && data[0] <= 16) return OP.OP_1 + data[0] - 1
  if (data.length === 1 && data[0] === 0x81) return OP.OP_1NEGATE
  if (data.length <= 75) return data.length
  if (data.length <= 0xff) return OP.OP_PUSHDATA1
  if (data.length <= 0xffff) return OP.OP_PUSHDATA2
  return OP.OP_PUSHDATA4
}
const chunkBytes = (c: ScriptChunk): number[] => {
  if (c.op === OP.OP_1NEGATE) return [0x81]
  if (c.op >= OP.OP_1 && c.op <= OP.OP_16) return [c.op - OP.OP_1 + 1]
  return c.data ?? []
}

const p2pkhOf = (rest: ScriptChunk[]): number[] | undefined =>
  rest.length === 5 && rest[0].op === OP.OP_DUP && rest[1].op === OP.OP_HASH160 &&
  rest[2].op === 20 && rest[2].data?.length === 20 && rest[3].op === OP.OP_EQUALVERIFY && rest[4].op === OP.OP_CHECKSIG
    ? [...rest[2].data] : undefined

export interface Bsv21BinaryDecoded { /* exactly as in Interfaces */ }

export class Bsv21Binary implements ScriptTemplate {
  constructor (private readonly wallet?: WalletInterface, private readonly originator?: string) {}

  static decode (script: LockingScript): Bsv21BinaryDecoded {
    if (!isTokenShaped(script)) fail('not a BRC-162 token output')
    const c = script.chunks
    if (c.some(ch => ch.invalidLength === true)) fail('truncated push')
    let tokenId: number[] | undefined
    if (c[0].op !== OP.OP_0) {
      if (c[0].op !== 0x20 || c[0].data?.length !== 32) fail('token id must be a direct 32-byte push')
      tokenId = [...c[0].data!]
    }
    const amount = decodeAmountChunk(c[1])
    let rest = c.slice(3)
    let payload: number[] | undefined
    let payloadCanonical = true
    if (rest.length >= 2 && isPushOp(rest[0].op) && rest[1].op === OP.OP_DROP) {
      payload = chunkBytes(rest[0])
      payloadCanonical = rest[0].op === canonicalPushOp(payload)
      rest = rest.slice(2)
    }
    const role: Bsv21Role = tokenId === undefined ? 'deploy' : amount === 0n ? 'authority' : 'value'
    return { role, tokenId, amount, payload, payloadCanonical, restChunks: rest, restPubKeyHash: p2pkhOf(rest) }
  }

  lock (tokenId: string | null, amount: bigint, pubKeyHash: readonly number[], payload?: readonly number[]): LockingScript {
    if (pubKeyHash.length !== 20) fail('pubKeyHash must be 20 bytes')
    const chunks: ScriptChunk[] = [
      tokenId === null ? { op: OP.OP_0 } : { op: 0x20, data: tokenIdFromString(tokenId) },
      encodeAmountChunk(amount),
      { op: OP.OP_2DROP }
    ]
    if (payload !== undefined) {
      const op = canonicalPushOp(payload)
      chunks.push(op >= OP.OP_1 || op === OP.OP_0 || op === OP.OP_1NEGATE ? { op } : { op, data: [...payload] }, { op: OP.OP_DROP })
    }
    chunks.push({ op: OP.OP_DUP }, { op: OP.OP_HASH160 }, { op: 20, data: [...pubKeyHash] }, { op: OP.OP_EQUALVERIFY }, { op: OP.OP_CHECKSIG })
    return new LockingScript(chunks)
  }

  async lockBRC29 (tokenId: string | null, amount: bigint, protocolID: WalletProtocol, keyID: string, counterparty: WalletCounterparty, payload?: readonly number[]): Promise<LockingScript> {
    if (this.wallet === undefined) fail('lockBRC29 requires a wallet')
    const { publicKey } = await this.wallet!.getPublicKey({ protocolID, keyID, counterparty }, this.originator)
    return this.lock(tokenId, amount, hash160(toArray(publicKey, 'hex')), payload)
  }

  unlock (privateKey: PrivateKey, signOutputs: 'all' | 'none' | 'single' = 'all', anyoneCanPay = false): ScriptTemplateUnlock {
    // The sighash subscript is the source output's full locking script, so a P2PKH
    // unlocker spends the prefixed output unchanged.
    return new P2PKH().unlock(privateKey, signOutputs, anyoneCanPay)
  }
}
```
